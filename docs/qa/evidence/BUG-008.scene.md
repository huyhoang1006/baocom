💥 ROOT CAUSE: `next.config.ts` không có `headers()` async function
   để inject security headers vào mọi response.

🔧 FIX (thêm vào next.config.ts):
```typescript
import type { NextConfig } from "next";

const nextConfig: NextConfig = {
  allowedDevOrigins: ["192.168.0.102", "192.168.4.98"],

  async headers() {
    return [
      {
        source: '/(.*)',
        headers: [
          // Ngăn MIME sniffing
          { key: 'X-Content-Type-Options', value: 'nosniff' },
          // Chống clickjacking
          { key: 'X-Frame-Options', value: 'DENY' },
          // HTTPS only (production)
          { key: 'Strict-Transport-Security',
            value: 'max-age=31536000; includeSubDomains' },
          // Referer policy
          { key: 'Referrer-Policy',
            value: 'strict-origin-when-cross-origin' },
          // Permissions
          { key: 'Permissions-Policy',
            value: 'camera=(), microphone=(), geolocation=()' },
          // Tắt X-Powered-By
          { key: 'X-Powered-By', value: '' },
          // CSP (test kỹ — có thể break Next.js inline scripts)
          // { key: 'Content-Security-Policy',
          //   value: "default-src 'self'; img-src 'self' data: https://fonts.gstatic.com; font-src 'self' https://fonts.gstatic.com; style-src 'self' 'unsafe-inline'; script-src 'self' 'unsafe-inline' 'unsafe-eval'" },
        ],
      },
    ]
  },
};

export default nextConfig;
```
