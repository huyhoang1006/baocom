  1. Attacker gửi liên tục POST /api/auth/login

  3. Mỗi attempt chỉ tốn ~15-50ms

💥 ROOT CAUSE: Không có rate-limit middleware trong middleware.ts
   cũng như trong route handler. Env flag `RATE_LIMIT_BYPASS=true`
   tồn tại nhưng implementation bị thiếu.

🔧 FIX Option A: In-memory rate-limit (đơn giản, single-instance)
```typescript
// middleware.ts — thêm cho /api/auth/login
const loginAttempts = new Map<string, number[]>()

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl

  // Rate-limit cho /api/auth/login
  if (pathname === '/api/auth/login' && request.method === 'POST') {
    const ip = request.headers.get('x-forwarded-for') || 'unknown'
    const now = Date.now()
    const window = 60_000  // 1 phút
    const max = 10         // 10 attempts/phút/IP

    const attempts = (loginAttempts.get(ip) || [])
      .filter(t => now - t < window)
    attempts.push(now)
    loginAttempts.set(ip, attempts)

    if (attempts.length > max) {
      return NextResponse.json(
        { error: 'Too many login attempts. Please try again later.' },
        { status: 429, headers: { 'Retry-After': '60' } }
      )
    }
  }

  // ... existing /admin/* logic
}

export const config = {
  matcher: ['/admin/:path*', '/api/auth/login'],
}
```

🔧 FIX Option B: Production-grade với Redis (distributed)
```typescript
import { Ratelimit } from "@upstash/ratelimit"
import { Redis } from "@upstash/redis"

const limiter = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(10, "1 m"),
})

export const POST = async (req) => {
  const ip = req.headers.get("x-forwarded-for") ?? "unknown"
  const { success } = await limiter.limit(ip)
  if (!success) return new Response("Too many requests", { status: 429 })
  // ... login logic
}
```
